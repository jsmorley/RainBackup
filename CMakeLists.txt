cmake_minimum_required(VERSION 3.21)
project(RainBackup VERSION 2.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(Z third_party/zlib)
add_library(zip STATIC
    ${Z}/adler32.c ${Z}/crc32.c ${Z}/deflate.c ${Z}/inflate.c
    ${Z}/inffast.c ${Z}/inftrees.c ${Z}/trees.c ${Z}/zutil.c
    ${Z}/minizip/zip.c ${Z}/minizip/unzip.c ${Z}/minizip/ioapi.c ${Z}/minizip/iowin32.c)
target_include_directories(zip PUBLIC ${Z} ${Z}/minizip)
target_compile_definitions(zip PRIVATE NOCRYPT NOUNCRYPT _CRT_SECURE_NO_WARNINGS)
add_library(backup STATIC src/Model.cpp src/Archive.cpp src/Schedule.cpp)
target_include_directories(backup PUBLIC src)
target_compile_definitions(backup PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(backup PUBLIC zip shell32 ole32 oleaut32 advapi32 uuid taskschd)
add_executable(RainBackup WIN32 src/Main.cpp src/CommandLine.cpp src/RainmeterSession.cpp src/SetupSession.cpp src/RainBackup.rc)
target_link_libraries(RainBackup PRIVATE backup comctl32 dwmapi uxtheme)
target_compile_definitions(RainBackup PRIVATE RAINBACKUP_VERSION=L\"${PROJECT_VERSION}\")
if(MSVC)
    target_link_options(RainBackup PRIVATE /MANIFEST:NO)
    foreach(target backup RainBackup)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    endforeach()
endif()
enable_testing()
add_executable(BackupTests tests/BackupTests.cpp)
target_link_libraries(BackupTests PRIVATE backup)
add_test(NAME BackupRoundTrip COMMAND BackupTests)
